  <div class="result" id="result"></div>


  <script>

    const items = ['Úkol 1', 'Úkol 2', 'Úkol 3', 'Úkol 4', 'Úkol 5', 'Úkol 6'];

    const colors = ['#f43f5e', '#8b5cf6', '#0ea5e9', '#10b981', '#f59e0b', '#ec4899'];

    

    const canvas = document.getElementById('wheel');

    const ctx = canvas.getContext('2d');

    const spinBtn = document.getElementById('spin-btn');

    const resultDiv = document.getElementById('result');


    const numItems = items.length;

    const arcSize = (2 * Math.PI) / numItems;

    let currentRotation = 0;

    let isSpinning = false;


    // Vykreslení kola

    function drawWheel() {

      const centerX = canvas.width / 2;

      const centerY = canvas.height / 2;

      const radius = canvas.width / 2 - 10;


      ctx.clearRect(0, 0, canvas.width, canvas.height);


      for (let i = 0; i < numItems; i++) {

        const angle = i * arcSize;

        

        // Výseč

        ctx.beginPath();

        ctx.fillStyle = colors[i % colors.length];

        ctx.moveTo(centerX, centerY);

        ctx.arc(centerX, centerY, radius, angle, angle + arcSize);

        ctx.lineTo(centerX, centerY);

        ctx.fill();

        ctx.stroke();


        // Text

        ctx.save();

        ctx.fillStyle = '#ffffff';

        ctx.font = 'bold 18px sans-serif';

        ctx.translate(centerX, centerY);

        ctx.rotate(angle + arcSize / 2);

        ctx.textAlign = 'right';

        ctx.fillText(items[i], radius - 20, 6);

        ctx.restore();

      }

    }


    // Zatočení kolem

    spinBtn.addEventListener('click', () => {

      if (isSpinning) return;

      

      isSpinning = true;

      spinBtn.disabled = true;

      resultDiv.textContent = '';


      // Náhodné otočení (min 5 plných otáček + náhodný úhel)

      const extraDegrees = Math.floor(Math.random() * 360) + 1800;

      currentRotation += extraDegrees;


      canvas.style.transform = `rotate(${currentRotation}deg)`;


      // Vyhodnocení výhry po dokončení animace (4s)

      setTimeout(() => {

        isSpinning = false;

        spinBtn.disabled = false;


        const actualDegrees = currentRotation % 360;

        // Šipka ukazuje nahoru (270 stupňů v Canvas souřadnicích)

        const effectiveAngle = (360 - (actualDegrees % 360) + 270) % 360;

        const selectedIndex = Math.floor(effectiveAngle / (360 / numItems));


        resultDiv.textContent = `Padlo: ${items[selectedIndex]}`;

      }, 4000);

    });


    drawWheel();

  </script>
