    // ---------- Pomocné funkce ----------

    function todayStr() { return new Date().toISOString().slice(0, 10); }

    function daysBetween(a, b) { return Math.round((new Date(b + "T00:00:00") - new Date(a + "T00:00:00")) / 86400000); }

    function isSeasonActive(t, m) { return !t.months || t.months.length === 0 || t.months.includes(m); }

    function isDue(t, today) { return !t.lastDone || daysBetween(t.lastDone, today) >= t.interval; }

    

    function storageGet(k) { try { const r = localStorage.getItem(k); return r ? JSON.parse(r) : null; } catch(e) { return null; } }

    function storageSet(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch(e) {} }

        setSessions(storageGet("work-sessions") || []);

      const today = todayStr();

      const dueIndoor = due.filter(t => t.category === "indoor");

      function toggleTask(id) {

        const updated = tasks.map(t => t.id === id ? { ...t, lastDone: t.lastDone === today ? null : today } : t);

        setTasks(updated);

        storageSet("tasks-catalog", updated);

      }


      function toggleRain() {

        const next = { date: today, isRaining: !weather.isRaining };

        setWeather(next);

        storageSet("weather-state", next);

      }


      function addExtra(name) {

        if (!name.trim()) return;

        const updated = [...extra, { id: "e-" + Date.now(), name, done: false }];

        setExtra(updated);

        storageSet("extra-tasks", updated);

      }


      function toggleExtra(id) {

        const updated = extra.map(e => e.id === id ? { ...e, done: !e.done } : e);

        setExtra(updated);

        storageSet("extra-tasks", updated);

      }


      function deleteExtra(id) {

        const updated = extra.filter(e => e.id !== id);

        setExtra(updated);

        storageSet("extra-tasks", updated);

      }


      function addTask(newTask) {

        const updated = [...tasks, { ...newTask, id: "t-" + Date.now(), lastDone: null }];

        setTasks(updated);

        storageSet("tasks-catalog", updated);

      }


      function deleteTask(id) {

        const updated = tasks.filter(t => t.id !== id);

        setTasks(updated);

        storageSet("tasks-catalog", updated);

      }


      function addLogEntry(text) {

        if (!text.trim()) return;

        const entry = { id: "l-" + Date.now(), date: today, time: new Date().toLocaleTimeString("cs-CZ", { hour: "2-digit", minute: "2-digit" }), text };

        const updated = [entry, ...log];

        setLog(updated);

        storageSet("handover-log", updated);

      }


      function startShift(area) {

        if (sessions.some(s => !s.end)) return;

        const updated = [{ id: "s-" + Date.now(), date: today, start: new Date().toISOString(), end: null, area: area || "safran" }, ...sessions];

        setSessions(updated);

        storageSet("work-sessions", updated);

      }


      function endShift() {

        const idx = sessions.findIndex(s => !s.end);

        if (idx < 0) return;

        const now = new Date().toISOString();

        const s = sessions[idx];

        const hours = Math.round(((new Date(now) - new Date(s.start)) / 3600000) * 100) / 100;

        const rate = wageSettings.areaRates[s.area] || wageSettings.defaultRate;

        const updated = sessions.map((x, i) => i === idx ? { ...x, end: now, hours, rate, wage: Math.round(hours * rate) } : x);

        setSessions(updated);

        storageSet("work-sessions", updated);

      }


      function deleteSession(id) {

        const updated = sessions.filter(s => s.id !== id);

        setSessions(updated);

        storageSet("work-sessions", updated);

      }

      return e("div", { className: "space-y-4" },

        e("div", { className: "flex gap-2" },

          e("textarea", {

            value: text,

            onChange: e => setText(e.target.value),

            placeholder: "Zapsat poznámku z dnešní směny / stav objektu…",

          })

        ),

        e("button", {

          onClick: () => { onAddLog(text); setText(""); },

          className: "px-4 py-2 rounded-xl bg-[#F2A93B] text-black font-semibold text-sm w-full"

        }, "Uložit zápis do deníku"),

    function VykazView({ sessions, onStart, onEnd, onDelete, wageSettings, setWageSettings }) {

      const [selectedArea, setSelectedArea] = useState("safran");

      const active = sessions.find(s => !s.end);

      const totalWage = sessions.reduce((acc, s) => acc + (s.wage || 0), 0);

      const totalHours = Math.round(sessions.reduce((acc, s) => acc + (s.hours || 0), 0) * 100) / 100;


      return e("div", { className: "space-y-6" },

        // Ovládání směny

        e("div", { className: "p-4 rounded-xl bg-[#24282C] border border-[#3A3F44] text-center" },

          active ? e("div", { className: "space-y-3" },

            e("p", { className: "text-sm text-green-400 font-semibold" }, `⏱️ Probíhá směna od ${new Date(active.start).toLocaleTimeString("cs-CZ", { hour: "2-digit", minute: "2-digit" })}`),

            e("button", { onClick: onEnd, className: "w-full py-3 rounded-xl bg-red-500 text-white font-bold text-sm" }, "Ukončit směnu")

          ) : e("div", { className: "space-y-3" },

            e("select", { value: selectedArea, onChange: e => setSelectedArea(e.target.value), className: "w-full p-2.5 rounded-xl bg-[#2E3338] border border-[#3A3F44] text-sm outline-none mb-2" },

              AREAS.map(a => e("option", { key: a.id, value: a.id }, a.label))

            ),

            e("button", { onClick: () => onStart(selectedArea), className: "w-full py-3 rounded-xl bg-[#F2A93B] text-black font-bold text-sm" }, "Začít novou směnu")

          )

        ),


        // Celkový přehled

        e("div", { className: "grid grid-cols-2 gap-3" },

          e("div", { className: "p-3 rounded-xl bg-[#2E3338] border border-[#3A3F44] text-center" },

            e("p", { className: "text-xs text-gray-400" }, "Odpracováno celkem"),

            e("p", { className: "text-lg font-bold text-[#F2A93B]" }, `${totalHours} hod`)

          ),

          e("div", { className: "p-3 rounded-xl bg-[#2E3338] border border-[#3A3F44] text-center" },

            e("p", { className: "text-xs text-gray-400" }, "Celková odměna"),

            e("p", { className: "text-lg font-bold text-green-400" }, `${totalWage} Kč`)

          )

        ),


        // Výpis směn

        e("div", null,

          e("h2", { className: "text-xs font-bold uppercase tracking-wider text-gray-400 mb-3" }, "Výkaz odpracovaných směn"),

          sessions.map(s => e("div", { key: s.id, className: "flex justify-between items-center p-3 mb-2 rounded-xl bg-[#24282C] border border-[#3A3F44]" },

            e("div", null,

              e("p", { className: "text-sm font-semibold" }, `${s.date} ${s.end ? "" : "(probíhá)"}`),

              e("p", { className: "text-xs text-gray-400" }, `Objekt: ${(AREAS.find(a=>a.id===s.area)||{}).label || s.area}`),

              s.hours && e("p", { className: "text-xs text-green-400" }, `${s.hours} h × ${s.rate} Kč = ${s.wage} Kč`)

            ),

            s.end && e("button", { onClick: () => onDelete(s.id), className: "text-gray-500 hover:text-red-400 text-xs px-2" }, "Smazat")

          ))

        )

      );

    }
